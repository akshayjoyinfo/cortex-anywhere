  customers {
    uuid id PK
    text legal_name
  }
  tenants {
    uuid id PK
    uuid customer_id FK
    text tier "pooled or silo"
    text status "active, quarantined, offboarded"
    int run_retention_days "1 to 180"
  }
  agents {
    uuid id PK
    uuid tenant_id FK "RLS key"
    text name
    text_array tools "allow list"
  }
  memory_item {
    uuid id PK
    uuid tenant_id FK "RLS key, idx 1"
    text namespace "idx 1"
    vector embedding "1024 dims, HNSW idx 2"
    uuid run_id "logical"
    uuid agent_id "logical"
    text source "run, mcp, admin"
    text trust "user or retrieved"
  }
  memory_episode {
    uuid id PK
    uuid tenant_id FK "RLS key"
    uuid run_id "logical"
    uuid agent_id "logical"
  }
  audit_event {
    bigint id PK "identity"
    uuid tenant_id FK "RLS key"
    text kind
    jsonb detail "holds run_id"
  }

TableRLSPolicycortex_app may

cortex.current_tenant() reads app.tenant_id, which the app sets per transaction. No tenant set means zero rows; a bad value is an error.
